# Arquivo de exemplo para configuração das API Keys
# Copie este arquivo para .env e configure suas chaves

# OpenAI API Key (obtenha em: https://platform.openai.com/api-keys)
OPENAI_API_KEY=sk-your-openai-api-key-here

# Google Gemini API Key (obtenha em: https://makersuite.google.com/app/apikey)
GEMINI_API_KEY=your-gemini-api-key-here

# Nota: Você pode configurar apenas uma das duas API Keys
# A interface mostrará apenas os provedores disponíveis
